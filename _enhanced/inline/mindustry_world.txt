    public void accept(int buffer, Item item){
        if(!accepts(buffer)) return;
        buffers[buffer][indexes[buffer]++] = BufferItem.get(item.id, Time.time);
    }

// DirectionalItemBuffer.java : public Item poll(int buffer, float speed)
    public Item poll(int buffer, float speed){
        if(indexes[buffer] > 0){
            long l = buffers[buffer][0];
            float time = BufferItem.time(l);

            if(Time.time >= time + speed || Time.time < time){
                return content.item(BufferItem.item(l));
            }
        }
        return null;
    }

// DirectionalItemBuffer.java : public void remove(int buffer)
    public void remove(int buffer){
        System.arraycopy(buffers[buffer], 1, buffers[buffer], 0, indexes[buffer] - 1);
        indexes[buffer] --;
    }

// DirectionalItemBuffer.java : public void write(Writes write)
    public void write(Writes write){
        for(int i = 0; i < 4; i++){
            write.b(indexes[i]);
            write.b(buffers[i].length);
            for(long l : buffers[i]){
                write.l(l);
            }
        }
    }

// ItemBuffer.java : public Item poll(float speed)
    public Item poll(float speed){
        if(index > 0){
            long l = buffer[0];
            float time = TimeItem.time(l);

            if(Time.time >= time + speed || Time.time < time){
                return content.item(TimeItem.item(l));
            }
        }
        return null;
    }

// ItemBuffer.java : public void remove()
    public void remove(){
        System.arraycopy(buffer, 1, buffer, 0, index - 1);
        index--;
    }

// ItemBuffer.java : public void write(Writes write)
    public void write(Writes write){
        write.b((byte)index);
        write.b((byte)buffer.length);
        for(long l : buffer){
            write.l(l);
        }
    }

// ItemBuffer.java : public void read(Reads read)
    public void read(Reads read){
        index = read.b();
        byte length = read.b();
        for(int i = 0; i < length; i++){
            long l = read.l();
            if(i < buffer.length){
                buffer[i] = l;
            }
        }
        index = Math.min(index, length - 1);
    }
