// FxBufferQueue.java : public void dispose()
    @Override
    public void dispose(){
        for(int i = 0; i < buffers.size; i++){
            buffers.get(i).dispose();
        }
    }

// FxBufferQueue.java : public void resize(int width, int height)
    public void resize(int width, int height){
        for(int i = 0; i < buffers.size; i++){
            buffers.get(i).resize(width, height);
        }
    }

// FxBufferQueue.java : public void rebind()
    public void rebind(){
        for(int i = 0; i < buffers.size; i++){
            FrameBuffer wrapper = buffers.get(i);
            // FBOs might be null if the instance wasn't initialized with #resize(int, int) yet.
            if(wrapper == null) continue;

            Texture texture = wrapper.getTexture();
            texture.setWrap(wrapU, wrapV);
            texture.setFilter(filterMin, filterMag);
        }
    }

// FxBufferQueue.java : public FrameBuffer getCurrent()
    public FrameBuffer getCurrent(){
        return buffers.get(currentIdx);
    }

// FxBufferQueue.java : public FrameBuffer changeToNext()
    public FrameBuffer changeToNext(){
        currentIdx = (currentIdx + 1) % buffers.size;
        return getCurrent();
    }

// FxBufferQueue.java : public void setTextureParams(TextureWrap u, TextureWrap v, TextureFilter min, TextureFilter mag)
    public void setTextureParams(TextureWrap u, TextureWrap v, TextureFilter min, TextureFilter mag){
        wrapU = u;
        wrapV = v;
        filterMin = min;
        filterMag = mag;
        rebind();
    }

// FxWidgetGroup.java : private void initialize()
    private void initialize(){
        if(initialized) return;

        performPendingResize();

        resizePending = false;
        initialized = true;
    }

// FxWidgetGroup.java : private void reset()
    private void reset(){
        if(!initialized) return;

        fxProcessor.dispose();

        resizePending = false;
        initialized = false;
    }

// PingPongBuffer.java : public void begin()
    public void begin(){
        if(capturing){
            throw new IllegalStateException("Ping pong buffer is already in capturing state.");
        }

        capturing = true;
        bufDst.begin();
    }

// PingPongBuffer.java : public void end()
    public void end(){
        if(!capturing){
            throw new IllegalStateException("Ping pong is not in capturing state. You should call begin() before calling end().");
        }
        bufDst.end();
        capturing = false;
    }
