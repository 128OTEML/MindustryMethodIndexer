// UnlockableContent.java : public void unlock()
    public void unlock(){
        if(!unlocked && !alwaysUnlocked){
            unlocked = true;
            Core.settings.put(name + "-unlocked", true);

            onUnlock();
            Events.fire(new UnlockEvent(this));
        }
    }

// UnlockableContent.java : public void quietUnlock()
    public void quietUnlock(){
        if(!unlocked()){
            unlocked = true;
            Core.settings.put(name + "-unlocked", true);
        }
    }

// UnlockableContent.java : public boolean unlockedNowHost()
    public boolean unlockedNowHost(){
        return !state.isCampaign() || unlockedHost();
    }

// UnlockableContent.java : public boolean unlockedHost()
    public boolean unlockedHost(){
        return net != null && net.client() ?
            alwaysUnlocked || state.rules.researched.contains(this) :
            unlocked || alwaysUnlocked;
    }

// UnlockableContent.java : public boolean unlockedNow()
    public boolean unlockedNow(){
        return unlocked() || !state.isCampaign();
    }

// UnlockableContent.java : public boolean unlocked()
    public boolean unlocked(){
        return net != null && net.client() ?
            alwaysUnlocked || unlocked || state.rules.researched.contains(this) :
            unlocked || alwaysUnlocked;
    }

// UnlockableContent.java : public void clearUnlock()
    public void clearUnlock(){
        if(unlocked){
            unlocked = false;
            Core.settings.put(name + "-unlocked", false);
        }
    }

// UnlockableContent.java : public boolean locked()
    public boolean locked(){
        return !unlocked();
    }
