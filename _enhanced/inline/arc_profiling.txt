// GL20Interceptor.java : private void check()
    private void check(){
        if(!Core.app.isOnMainThread()){
            glProfiler.getListener().onError("GL call on wrong thread: " + Thread.currentThread());
        }
        int error = gl20.glGetError();
        while(error != GL20.GL_NO_ERROR){
            glProfiler.getListener().onError(resolveErrorNumber(error));
            error = gl20.glGetError();
        }
    }

// GL30Interceptor.java : private void check()
    private void check(){
        if(!Core.app.isOnMainThread()){
            glProfiler.getListener().onError("GL call on wrong thread: " + Thread.currentThread());
        }
        int error = gl30.glGetError();
        while(error != GL20.GL_NO_ERROR){
            glProfiler.getListener().onError(resolveErrorNumber(error));
            error = gl30.glGetError();
        }
    }

// GLInterceptor.java : public static String resolveErrorNumber(int error)
    public static String resolveErrorNumber(int error){
        switch(error){
            case GL_INVALID_VALUE: return "GL_INVALID_VALUE";
            case GL_INVALID_OPERATION: return "GL_INVALID_OPERATION";
            case GL_INVALID_FRAMEBUFFER_OPERATION: return "GL_INVALID_FRAMEBUFFER_OPERATION";
            case GL_INVALID_ENUM: return "GL_INVALID_ENUM";
            case GL_OUT_OF_MEMORY: return "GL_OUT_OF_MEMORY";
            default: return "number " + error;
        }
    }

// GLInterceptor.java : public void reset()
    public void reset(){
        calls = 0;
        textureBindings = 0;
        drawCalls = 0;
        shaderSwitches = 0;
        stateChanges = 0;
        vertexCount.reset();
    }

// GLProfiler.java : public void disable()
    public void disable(){
        if(!enabled) return;

        GL30 gl30 = graphics.getGL30();
        if(gl30 != null) graphics.setGL30(((GL30Interceptor)graphics.getGL30()).gl30);
        else graphics.setGL20(((GL20Interceptor)graphics.getGL20()).gl20);

        enabled = false;
    }
