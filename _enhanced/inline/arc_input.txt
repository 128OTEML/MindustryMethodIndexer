// InputEventQueue.java : private void queueTime()
    private void queueTime(){
        long time = Time.nanos();
        queue.add((int)(time >> 32));
        queue.add((int)time);
    }

// InputEventQueue.java : public synchronized boolean keyDown(KeyCode key)
    @Override
    public synchronized boolean keyDown(KeyCode key){
        queue.add(KEY_DOWN);
        queueTime();
        queue.add(key.ordinal());
        return false;
    }

// InputMultiplexer.java : public boolean keyDown(KeyCode keycode)
    @Override
    public boolean keyDown(KeyCode keycode){
        Object[] items = processors.begin();
        try{
            for(int i = 0, n = processors.size; i < n; i++)
                if(((InputProcessor)items[i]).keyDown(keycode)) return true;
        }finally{
            processors.end();
        }
        return false;
    }

// KeyBind.java : public void save()
    public void save(){
        String name = settingsKey();
        settings.put(name + "-single", value.key != null);

        if(value.key != null){
            settings.put(name + "-key", value.key.ordinal());
        }else{
            settings.put(name + "-min", value.min.ordinal());
            settings.put(name + "-max", value.max.ordinal());
        }
    }

// KeyBind.java : public void unset()
    public void unset(){
        value = new Axis(KeyCode.unset);
    }

// KeyBind.java : public boolean isDefault()
    public boolean isDefault(){
        if(defaultValue instanceof Axis){
            if(((Axis)defaultValue).min == null){
                return ((Axis)defaultValue).key == value.key;
            }else{
                return ((Axis)defaultValue).max == value.max && ((Axis)defaultValue).min == value.min;
            }
        }else{
            return defaultValue == value.key;
        }
    }

// KeyBind.java : public boolean isUnset()
    public boolean isUnset(){
        return value.key == KeyCode.unset;
    }

// KeyBind.java : String settingsKey()
    String settingsKey(){
        return "keybind-default-keyboard-" + name;
    }

    }

// KeyboardDevice.java : public boolean isPressed(KeyCode key)
    public boolean isPressed(KeyCode key){
        if(key == KeyCode.anyKey) return pressed.size > 0;

        return pressed.contains(key.ordinal());
    }

        return false;
    }
