// SectorInfo.java : void updateDelta(Item item, ObjectMap<Item,ExportStat> map, int[] deltas)
    void updateDelta(Item item, ObjectMap<Item, ExportStat> map, int[] deltas){
        ExportStat stat = map.get(item, ExportStat::new);
        if(!stat.loaded){
            stat.means.fill(stat.mean);
            stat.loaded = true;
        }

        //store means
        stat.means.add(deltas[item.id]);
        stat.mean = stat.means.rawMean();
    }

// SpawnGroup.java : public int getSpawned(int wave)
    public int getSpawned(int wave){
        if(spacing == 0) spacing = 1;
        if(wave < begin || wave > end || (wave - begin) % spacing != 0){
            return 0;
        }
        return Math.min(unitAmount + (int)(((wave - begin) / spacing) / unitScaling), max);
    }

// SpawnGroup.java : public float getShield(int wave)
    public float getShield(int wave){
        return Math.max(shields + shieldScaling*(wave - begin), 0);
    }

// SpawnGroup.java : public boolean equals(Object o)
    @Override
    public boolean equals(Object o){
        if(this == o) return true;
        if(o == null || getClass() != o.getClass()) return false;
        SpawnGroup group = (SpawnGroup)o;
        return end == group.end && begin == group.begin && spacing == group.spacing && max == group.max
            && Float.compare(group.unitScaling, unitScaling) == 0 && Float.compare(group.shields, shields) == 0
            && Float.compare(group.shieldScaling, shieldScaling) == 0 && unitAmount == group.unitAmount &&
            type == group.type && effect == group.effect && Structs.eq(items, group.items);
    }

// Universe.java : private void updatePlanet(Planet planet)
    private void updatePlanet(Planet planet){
        planet.position.setZero();
        planet.addParentOffset(planet.position);
        if(planet.parent != null){
            planet.position.add(planet.parent.position);
        }
        for(Planet child : planet.children){
            updatePlanet(child);
        }
    }
