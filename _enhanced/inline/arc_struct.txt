    public E poll(){
        if(size == 0) return null;
        int s = --size;
        E result = (E)queue[0];
        E x = (E)queue[s];
        queue[s] = null;
        if(s != 0) siftDown(0, x);
        return result;
    }

// PQueue.java : private void siftUp(int k, E x)
    private void siftUp(int k, E x){
        while(k > 0){
            int parent = (k - 1) >>> 1;
            E e = (E)queue[parent];
            if(compare(x, e) >= 0) break;
            queue[k] = e;
            k = parent;
        }
        queue[k] = x;
    }

// SnapshotSeq.java : public void end()
    public void end(){
        snapshots = Math.max(0, snapshots - 1);
        if(snapshot == null) return;
        if(snapshot != items && snapshots == 0){
            // The backing array was copied, keep around the old array.
            recycled = snapshot;
            for(int i = 0, n = recycled.length; i < n; i++)
                recycled[i] = null;
        }
        snapshot = null;
    }

// SnapshotSeq.java : private void modified()
    private void modified(){
        if(snapshot == null || snapshot != items) return;
        // Snapshot is in use, copy backing array to recycled array or create new backing array.
        if(recycled != null && recycled.length >= size){
            System.arraycopy(items, 0, recycled, 0, size);
            items = recycled;
            recycled = null;
        }else
            resize(items.length);
    }

    private static int minRunLength(int n){

// TimSort.java : private void mergeCollapse()
    private void mergeCollapse(){
        while(stackSize > 1){
            int n = stackSize - 2;
            if((n >= 1 && runLen[n - 1] <= runLen[n] + runLen[n + 1]) || (n >= 2 && runLen[n - 2] <= runLen[n] + runLen[n - 1])){
                if(runLen[n - 1] < runLen[n + 1]) n--;
            }else if(runLen[n] > runLen[n + 1]){
                break; // Invariant is established
            }
            mergeAt(n);
        }
    }

        }
