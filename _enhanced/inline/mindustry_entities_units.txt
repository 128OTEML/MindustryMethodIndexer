// AIController.java : public void faceTarget()
    public void faceTarget(){
        if(unit.type.omniMovement || unit instanceof Mechc){
            if(!Units.invalidateTarget(target, unit, unit.range()) && unit.type.faceTarget && unit.type.hasWeapons()){
                unit.lookAt(Predict.intercept(unit, target, unit.type.weapons.first().bullet));
            }else if(unit.moving()){
                unit.lookAt(unit.vel().angle());
            }
        }
    }

// BuildPlan.java : public boolean isRotation(Team team)
    public boolean isRotation(Team team){
        if(breaking) return false;
        Tile tile = tile();
        return tile != null && tile.team() == team && tile.block() == block && tile.build != null && tile.build.rotation != rotation;
    }

// BuildPlan.java : public boolean isDerelictRepair()
    public boolean isDerelictRepair(){
        if(breaking || !state.rules.derelictRepair) return false;
        Tile tile = tile();
        return tile != null && tile.team() == Team.derelict && tile.block() == block && tile.build != null;
    }

// BuildPlan.java : public BuildPlan set(int x, int y, int rotation, Block block)
    public BuildPlan set(int x, int y, int rotation, Block block){
        this.x = x;
        this.y = y;
        if(block != null) this.rotation = block.planRotation(rotation);
        this.block = block;
        this.breaking = false;
        return this;
    }

// BuildPlan.java : public float drawx()
    public float drawx(){
        return x*tilesize + (block == null ? 0 : block.offset);
    }

// BuildPlan.java : public float drawy()
    public float drawy(){
        return y*tilesize + (block == null ? 0 : block.offset);
    }

// BuildPlan.java : public boolean isDone()
    public boolean isDone(){
        Tile tile = world.tile(x, y);
        if(tile == null) return true;
        Block tblock = tile.block();
        if(breaking){
            return tblock == Blocks.air || tblock == tile.floor();
        }else{
            return tblock == block && (tile.build == null || tile.build.rotation == rotation);
        }
    }

// BuildPlan.java : public Tile tile()
    public @Nullable Tile tile(){
        return world.tile(x, y);
    }

    }

// BuildPlan.java : public void hitbox(Rect out)
    @Override
    public void hitbox(Rect out){
        if(block != null){
            out.setCentered(x * tilesize + block.offset, y * tilesize + block.offset, block.size * tilesize);
        }else{
            out.setCentered(x * tilesize, y * tilesize, tilesize);
        }
    }
