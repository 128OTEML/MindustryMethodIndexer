// Administration.java : public boolean banPlayerID(String id)
    public boolean banPlayerID(String id){

            return false;

        save();

        return true;
    }

// Administration.java : public boolean unbanPlayerID(String id)
    public boolean unbanPlayerID(String id){

        if(!info.banned) return false;

        info.banned = false;
        bannedIPs.removeAll(info.ips, false);
        save();
        Events.fire(new PlayerUnbanEvent(Groups.player.find(p -> id.equals(p.uuid())), id));
        return true;
    }

        }
    }

// NetworkIO.java : private static void writeString(ByteBuffer buffer, String string, int maxlen) / private static void writeString(ByteBuffer buffer, String string)
    private static void writeString(ByteBuffer buffer, String string, int maxlen){
        byte[] bytes = string.getBytes(charset);
        //todo truncating this way may lead to wierd encoding errors at the ends of strings...
        if(bytes.length > maxlen){
            bytes = Arrays.copyOfRange(bytes, 0, maxlen);
        }

        buffer.put((byte)bytes.length);
        buffer.put(bytes);
    }

// NetworkIO.java : private static String readString(ByteBuffer buffer)
    private static String readString(ByteBuffer buffer){
        short length = (short)(buffer.get() & 0xff);
        byte[] bytes = new byte[length];
        buffer.get(bytes);
        return new String(bytes, charset);
    }
