    private Page getBest(Page result1, Page result2){

        if(result2 == null) return result1;

    }

// TexturePacker.java : public void setRootDir(File rootDir)
    public void setRootDir(File rootDir){
        if(rootDir == null){
            rootPath = null;
            return;
        }
        rootPath = rootDir.getAbsolutePath().replace('\\', '/');
        if(!rootPath.endsWith("/")) rootPath += "/";
    }

// TexturePacker.java : private static void copy(Pixmap src, int x, int y, int w, int h, Pixmap dst, int dx, int dy, boolean rotated)
    private static void copy(Pixmap src, int x, int y, int w, int h, Pixmap dst, int dx, int dy, boolean rotated){
        if(rotated){
            for(int i = 0; i < w; i++)
                for(int j = 0; j < h; j++)
                    dst.set(dx + j, dy + w - i - 1, src.getRaw(x + i, y + j));
        }else{
            dst.draw(src, x, y, w, h, dx, dy, w, h);
        }
    }

// TexturePacker.java : public static boolean processIfModified(String input, String output, String packFileName) / public static boolean processIfModified(Settings settings, String input, String output, String packFileName)
    public static boolean processIfModified(String input, String output, String packFileName){
        // Default settings (Needed to access the default atlas extension string)
        Settings settings = new Settings();

        if(isModified(input, output, packFileName, settings)){
            process(settings, input, output, packFileName);
            return true;
        }
        return false;
    }

// TextureUnpacker.java : private boolean checkDirectoryValidity(String directory)
    private boolean checkDirectoryValidity(String directory){
        File checkFile = new File(directory);
        boolean path = true;
        // try to get the canonical path, if this fails the path is not valid
        try{
            checkFile.getCanonicalPath();
        }catch(Exception e){
            path = false;
        }
        return path;
    }
