// Ability.java : public void display(Table t)
    public void display(Table t){
        t.table(Styles.grayPanel, a -> {
            a.add("[accent]" + localized()).padBottom(4).center().top().expandX();
            a.row();
            a.left().top().defaults().left();
            addStats(a);
        }).pad(5).margin(10).growX().top().uniformX();
    }

// Ability.java : public void addStats(Table t)
    public void addStats(Table t){
        if(Core.bundle.has(getBundle() + ".description")){
            t.add(Core.bundle.get(getBundle() + ".description")).wrap().width(descriptionWidth);
            t.row();
        }
    }

// Ability.java : public String abilityStat(String stat, Object values)
    public String abilityStat(String stat, Object... values){
        return Core.bundle.format("ability.stat." + stat, values);
    }

// Ability.java : public Ability copy()
    public Ability copy(){
        try{
            return (Ability)clone();
        }catch(CloneNotSupportedException e){
            //I am disgusted
            throw new RuntimeException("java sucks", e);
        }
    }

// Ability.java : public String localized()
    public String localized(){
        return Core.bundle.get(getBundle());
    }

// Ability.java : public String getBundle()
    public String getBundle(){
        var type = getClass();
        return "ability." + (type.isAnonymousClass() ? type.getSuperclass() : type).getSimpleName().replace("Ability", "").toLowerCase();
    }

// ArmorPlateAbility.java : public void update(Unit unit)
    @Override
    public void update(Unit unit){
        super.update(unit);

        warmup = Mathf.lerpDelta(warmup, unit.isShooting() ? 1f : 0f, 0.1f);
        unit.healthMultiplier += warmup * healthMultiplier;
    }

// ArmorPlateAbility.java : public void addStats(Table t)
    @Override
    public void addStats(Table t){
        super.addStats(t);
        t.add(abilityStat("damagereduction", Strings.autoFixed(-healthMultiplier * 100f, 1)));
    }

// ForceFieldAbility.java : public float scaledMax(Unit unit)
    public float scaledMax(Unit unit){

    }

// ForceFieldAbility.java : public void death(Unit unit)
    @Override
    public void death(Unit unit){

        //self-destructing units can have a shield on death
        if(unit.shield > 0f && !wasBroken){
            Fx.shieldBreak.at(unit.x, unit.y, radius, unit.type.shieldColor(unit), sides);
            breakSound.at(unit.x, unit.y);
        }
    }

// ForceFieldAbility.java : public void displayBars(Unit unit, Table bars)
    @Override
    public void displayBars(Unit unit, Table bars){
        bars.add(new Bar("stat.shieldhealth", Pal.accent, () -> unit.shield / scaledMax(unit))).row();
    }

// SuppressionFieldAbility.java : public void update(Unit unit)
    @Override
    public void update(Unit unit){
        if(!active) return;

        if((timer += Time.delta) >= maxDelay){
            Tmp.v1.set(x, y).rotate(unit.rotation - 90f).add(unit);
            Damage.applySuppression(unit.team, Tmp.v1.x, Tmp.v1.y, range, reload, maxDelay, applyParticleChance, unit, effectColor);
            timer = 0f;
        }
    }

// UnitSpawnAbility.java : public void draw(Unit unit)
    @Override
    public void draw(Unit unit){
        if(Units.canCreate(unit.team, this.unit)){
            Draw.draw(Draw.z(), () -> {
                float x = unit.x + Angles.trnsx(unit.rotation, spawnY, -spawnX), y = unit.y + Angles.trnsy(unit.rotation, spawnY, -spawnX);
                Drawf.construct(x, y, this.unit.fullIcon, unit.rotation - 90, timer / spawnTime, 1f, timer);
            });
        }
    }
