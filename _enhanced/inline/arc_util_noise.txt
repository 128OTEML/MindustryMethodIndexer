// Noise.java : private static void normalize2(double[] v)
    private static void normalize2(double[] v){
        double s;
        s = Math.sqrt(v[0] * v[0] + v[1] * v[1]);
        v[0] = v[0] / s;
        v[1] = v[1] / s;
    }

    static double range(double n){
        if(n >= 1073741824.0)
            return (2.0 * (n % 1073741824.0)) - 1073741824.0;
        else if(n <= -1073741824.0)
            return (2.0 * (n % 1073741824.0)) + 1073741824.0;
        else
            return n;
    }

    static double scurve5(double a){
        double a3 = a * a * a;
        double a4 = a3 * a;
        double a5 = a4 * a;
        return (6.0 * a5) - (15.0 * a4) + (10.0 * a3);
    }

// Simplex.java : static int perm(int seed, int x)
    static int perm(int seed, int x){
        x = (x & 255) * 0x45d9f3b;
        x = ((x >>> 16) ^ x) * (0x45d9f3b + seed);
        x = (x >>> 16) ^ x;
        return x & 0xff;
    }
