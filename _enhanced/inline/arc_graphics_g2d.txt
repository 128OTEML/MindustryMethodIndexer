    protected void load(FontData data){

            if(page == null) continue;
            for(Glyph glyph : page)

// GlyphLayout.java : private void adjustLastGlyph(FontData fontData, GlyphRun run)
    private void adjustLastGlyph(FontData fontData, GlyphRun run){
        Glyph last = run.glyphs.peek();
        if(last.fixedWidth) return;
        float width = (last.width + last.xoffset) * fontData.scaleX - fontData.padRight;
        run.width += width - run.xAdvances.peek(); // Can cause the run width to be > targetWidth, but the problem is minimal.
        run.xAdvances.set(run.xAdvances.size - 1, width);
    }

    public static void dashLine(float x1, float y1, float x2, float y2, int divisions){

        }

// SpriteCache.java : public void beginCache() / public void beginCache(int cacheID)
    public void beginCache(){
        if(drawing) throw new IllegalStateException("end must be called before beginCache");
        if(currentCache != null) throw new IllegalStateException("endCache must be called before begin.");
        mesh.getVerticesBuffer().position(caches.isEmpty() ? 0 : caches.peek().offset + caches.peek().maxCount);
        currentCache = new Cache(caches.size, mesh.getVerticesBuffer().position());
        caches.add(currentCache);
        mesh.getVerticesBuffer().limit(mesh.getVerticesBuffer().capacity());
    }

    public void clear(){

    }
