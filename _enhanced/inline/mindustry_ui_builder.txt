// UiDslParser.java : public static TableBuilder parse(String source)
    public static TableBuilder parse(String source){
        TableBuilder root = new TableBuilder();
        UiDslParser p = new UiDslParser(source);
        p.parseStatementsInto(root);
        p.skipWs();
        if(!p.atEnd()) throw new SerializationException("Unexpected character '" + p.peek() + "' at line " + p.line(p.pos));
        return root;
    }

// UiDslParser.java : private String parseValue()
    private String parseValue(){
        skipWs();
        if(!atEnd() && peek() == '"'){
            lastQuoted = true;
            return parseString();
        }
        lastQuoted = false;
        String tok = readBareToken();
        if(tok.isEmpty()) throw new SerializationException("Expected value at line " + line(pos));
        return tok;
    }

// UiDslParser.java : private String readBareToken()
    private String readBareToken(){
        int start = pos;
        while(!atEnd()){
            char c = src.charAt(pos);
            if(Character.isWhitespace(c) || c == '{' || c == '}' || c == ':') break;
            pos++;
        }
        return src.substring(start, pos);
    }

// UiDslParser.java : private void skipWs()
    private void skipWs(){
        while(!atEnd()){
            char c = src.charAt(pos);
            if(Character.isWhitespace(c)){ pos++; continue; }
            if(c == '/' && pos + 1 < src.length() && src.charAt(pos + 1) == '/'){ // line comment
                while(!atEnd() && src.charAt(pos) != '\n') pos++;
                continue;
            }
            break;
        }
    }

// UiDslParser.java : private void expect(char c)
    private void expect(char c){
        if(atEnd() || peek() != c) throw new SerializationException("Expected '" + c + "', found '" + (atEnd() ? "<eof>" : peek()) + "' at line " + line(pos));
        pos++;
    }

// UiDslParser.java : private boolean atEnd()
    private boolean atEnd(){
        return pos >= src.length();
    }

// UiDslParser.java : private char peek()
    private char peek(){
        return src.charAt(pos);
    }

// UiDslParser.java : private int line(int index)
    private int line(int index){
        int line = 1;
        int end = Math.min(index, src.length());
        for(int i = 0; i < end; i++){
            if(src.charAt(i) == '\n') line++;
        }
        return line;
    }

// UiDslWriter.java : private static String formatValue(Object value)
    private static String formatValue(Object value){
        if(value instanceof Boolean b) return b ? "true" : "false";
        if(value instanceof Float f) return formatFloat(f);
        if(value instanceof String s) return needsQuotes(s) ? quote(s) : s;
        throw new IllegalArgumentException("Unsupported value type: " + value);
    }

// UiDslWriter.java : private static String formatFloat(float f)
    private static String formatFloat(float f){
        if(f == (long)f) return Long.toString((long)f);
        return Float.toString(f);
    }

// UiStyleLookup.java : public static T get(Class<T> type, String name)
    public static <T> @Nullable T get(Class<T> type, String name){
        if(styles.isEmpty()) init();
        return (T)styles.get(type, ObjectMap::new).get(name);
    }

// UiStyleLookup.java : private static void init()
    private static void init(){
        for(var field : Styles.class.getFields()){
            try{
                styles.get(field.getType(), ObjectMap::new).put(field.getName(), field.get(null));
            }catch(Exception e){
                Log.err(e);
            }
        }
    }

// UiTreeBuilder.java : private static boolean applyStyle(NodeBuilder<?> node, Class<S> styleType, Cons<S> setter)
    private static <S> boolean applyStyle(NodeBuilder<?> node, Class<S> styleType, Cons<S> setter){
        String name = node.str(UiKey.style);
        if(name == null) return false;
        S style = UiStyleLookup.get(styleType, name);
        if(style != null){
            setter.get(style);
            return true;
        }
        return false;
    }

        }
    }
