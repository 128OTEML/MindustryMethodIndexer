    float rawHeight(Vec3 position){

    }

// ErekirPlanetGenerator.java : float rawTemp(Vec3 position)
    float rawTemp(Vec3 position){
        return position.dst(0, 0, 1)*2.2f - Simplex.noise3d(seed, 8, 0.54f, 1.4f, 10f + position.x, 10f + position.y, 10f + position.z) * 2.9f;
    }

// SerpuloPlanetGenerator.java : float rawHeight(Vec3 position)
    float rawHeight(Vec3 position){

    }

// SerpuloPlanetGenerator.java : public boolean allowNumberedLaunch(Sector s)
    public boolean allowNumberedLaunch(Sector s){
        return s.hasBase() && !s.isAttacked() && (s.info.bestCoreType.size >= 4 || s.isBeingPlayed() && state.rules.defaultTeam.cores().contains(b -> b.block.size >= 4));
    }

// SerpuloPlanetGenerator.java : public boolean allowLanding(Sector sector)
    @Override
    public boolean allowLanding(Sector sector){
        return sector.planet.allowLaunchToNumbered && (sector.hasBase() || sector.near().contains(this::allowNumberedLaunch));
    }

// SerpuloPlanetGenerator.java : public void getColor(Vec3 position, Color out)
    @Override
    public void getColor(Vec3 position, Color out){
        Block block = getBlock(position, true);
        //replace salt with sand color
        if(block == Blocks.salt) block = Blocks.sand;
        out.set(block.mapColor).a(1f - block.albedo);
    }

// SerpuloPlanetGenerator.java : public void genTile(Vec3 position, TileGen tile)
    @Override
    public void genTile(Vec3 position, TileGen tile){
        tile.floor = getBlock(position, false);
        if(tile.floor == Blocks.darkPanel6) tile.floor = Blocks.darkPanel3;
        tile.block = tile.floor.asFloor().wall;

        if(Ridged.noise3d(seed + 1, position.x, position.y, position.z, 2, 22) > 0.31){
            tile.block = Blocks.air;
        }
    }

// SerpuloPlanetGenerator.java : protected float noise(float x, float y, double octaves, double falloff, double scl, double mag)
    @Override
    protected float noise(float x, float y, double octaves, double falloff, double scl, double mag){
        Vec3 v = sector.rect.project(x, y).scl(5f);
        return Simplex.noise3d(seed, octaves, falloff, 1f / scl, v.x, v.y, v.z) * (float)mag;
    }

// SerpuloPlanetGenerator.java : public void postGenerate(Tiles tiles)
    @Override
    public void postGenerate(Tiles tiles){
        if(sector.hasEnemyBase()){
            basegen.postGenerate();

            //spawn air enemies
            if(spawner.countGroundSpawns() == 0){
                state.rules.spawns = Waves.generate(sector.threat, new Rand(sector.id), state.rules.attackMode, true, false);
            }
        }
    }
