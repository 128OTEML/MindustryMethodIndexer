// FxProcessor.java : public void setEffectPriority(FxFilter effect, int priority)
    public void setEffectPriority(FxFilter effect, int priority){
        priorities.put(effect, priority);
        effectsAll.sort(e -> priorities.get(effect, 0));
    }

// FxProcessor.java : public boolean end()
    public boolean end(){
        if(!capturing) return false;

        Draw.flush();

        hasCaptured = true;
        capturing = false;
        pingPongBuffer.end();
        return true;
    }
