    private <T extends Content> T find(ContentType type, String name){

        if(c == null && currentMod != null) c = Vars.content.getByName(type, currentMod.name + "-" + name);

        content.minfo.mod = mod;
        content.minfo.sourceFile = file;
        content.minfo.error = makeError(error, file);

// DataAssetCache.java : public synchronized byte[] add(byte[] bytes)
    public synchronized byte[] add(byte[] bytes){
        byte[] hash = Streams.sha256(bytes);
        String name = encodeHash(hash);
        Fi file = Vars.assetCacheDirectory.child(name);
        //avoid unnecessary disk writes when adding an asset that already exists. TODO: it's possible the file may be corrupted even if length matches?
        if(file.length() != bytes.length){
            file.writeBytes(bytes);
        }
        hashToFile.put(name, file);
        return hash;
    }

// DataPatcher.java : static Object copyArray(Object object)
    static Object copyArray(Object object){
        if(object instanceof int[] i) return i.clone();
        else if(object instanceof long[] i) return i.clone();
        else if(object instanceof short[] i) return i.clone();
        else if(object instanceof byte[] i) return i.clone();
        else if(object instanceof boolean[] i) return i.clone();
        else if(object instanceof char[] i) return i.clone();
        else if(object instanceof float[] i) return i.clone();
        else if(object instanceof double[] i) return i.clone();
        else return ((Object[])object).clone();
    }

// Mods.java : public Seq<LoadedMod> orderedMods()
    public Seq<LoadedMod> orderedMods(){
        //update cache if it's "dirty"/empty
        if(lastOrderedMods == null){
            //only enabled mods participate; this state is resolved in load()
            Seq<LoadedMod> enabled = mods.select(LoadedMod::enabled);

            var mapping = enabled.asMap(m -> m.meta.internalName);
            lastOrderedMods = resolveDependencies(enabled.map(m -> m.meta)).orderedKeys().map(mapping::get);
        }
        return lastOrderedMods;
    }

    public void setEnabled(LoadedMod mod, boolean enabled){
        if(mod.enabled() != enabled){
            Core.settings.put("mod-" + mod.name + "-enabled", enabled);
            Core.settings.put("mod-" + mod.name + "-failed", false);

            mods.each(this::updateDependencies);
            sortMods();
