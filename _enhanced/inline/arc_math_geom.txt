// Intersector3D.java : public static boolean intersectSegmentPlane(Vec3 start, Vec3 end, Plane plane, Vec3 intersection)
    public static boolean intersectSegmentPlane(Vec3 start, Vec3 end, Plane plane, Vec3 intersection){
        Vec3 dir = v0.set(end).sub(start);
        float denom = dir.dot(plane.normal);
        if(denom == 0f) return false;
        float t = -(start.dot(plane.normal) + plane.d) / denom;
        if(t < 0 || t > 1) return false;

        intersection.set(start).add(dir.scl(t));
        return true;
    }

        vec[1] = y;

// Quat.java : public int getGimbalPole()
    public int getGimbalPole(){
        final float t = y * x + z * w;
        return t > 0.499f ? 1 : (t < -0.499f ? -1 : 0);
    }

// Quat.java : public float getRollRad()
    public float getRollRad(){
        final int pole = getGimbalPole();
        return pole == 0 ? Mathf.atan2(1f - 2f * (x * x + z * z), 2f * (w * z + y * x)) : (float)pole * 2f
        * Mathf.atan2(w, y);
    }

// Quat.java : public float getRoll()
    public float getRoll(){
        return getRollRad() * Mathf.radiansToDegrees;
    }

// Quat.java : public float getPitchRad()
    public float getPitchRad(){
        final int pole = getGimbalPole();
        return pole == 0 ? (float)Math.asin(Mathf.clamp(2f * (w * x - z * y), -1f, 1f)) : (float)pole * Mathf.PI * 0.5f;
    }

// Quat.java : public float getPitch()
    public float getPitch(){
        return getPitchRad() * Mathf.radiansToDegrees;
    }

// Quat.java : public float getYawRad()
    public float getYawRad(){
        return getGimbalPole() == 0 ? Mathf.atan2(1f - 2f * (y * y + x * x), 2f * (y * w + x * z)) : 0f;
    }

// Quat.java : public float getYaw()
    public float getYaw(){
        return getYawRad() * Mathf.radiansToDegrees;
    }

// Quat.java : public Quat nor()
    public Quat nor(){
        float len = len2();
        if(len != 0.f && !Mathf.equal(len, 1f)){
            len = (float)Math.sqrt(len);
            w /= len;
            x /= len;
            y /= len;
            z /= len;
        }
        return this;
    }

// Quat.java : public Quat conjugate()
    public Quat conjugate(){
        x = -x;
        y = -y;
        z = -z;
        return this;
    }

// Quat.java : public Vec3 transform(Vec3 v)
    public Vec3 transform(Vec3 v){

// Bezier.java : public static T linear(T out, float t, T p0, T p1, T tmp)
    public static <T extends Vector<T>> T linear(final T out, final float t, final T p0, final T p1, final T tmp){
        // B1(t) = p0 + (p1-p0)*t
        return out.set(p0).scl(1f - t).add(tmp.set(p1).scl(t)); // Could just use lerp...
    }

// Bezier.java : public static T linearDerivative(T out, float t, T p0, T p1, T tmp)
    public static <T extends Vector<T>> T linearDerivative(final T out, final float t, final T p0, final T p1, final T tmp){
        // B1'(t) = p1-p0
        return out.set(p1).sub(p0);
    }

// Bezier.java : public static T quadratic(T out, float t, T p0, T p1, T p2, T tmp)
    public static <T extends Vector<T>> T quadratic(final T out, final float t, final T p0, final T p1, final T p2, final T tmp){
        // B2(t) = (1 - t) * (1 - t) * p0 + 2 * (1-t) * t * p1 + t*t*p2
        final float dt = 1f - t;
        return out.set(p0).scl(dt * dt).add(tmp.set(p1).scl(2 * dt * t)).add(tmp.set(p2).scl(t * t));
    }

// Bezier.java : public static T quadraticDerivative(T out, float t, T p0, T p1, T p2, T tmp)
    public static <T extends Vector<T>> T quadraticDerivative(final T out, final float t, final T p0, final T p1, final T p2,
                                                              final T tmp){
        // B2'(t) = 2 * (1 - t) * (p1 - p0) + 2 * t * (p2 - p1)
        final float dt = 1f - t;
        return out.set(p1).sub(p0).scl(2).scl(1 - t).add(tmp.set(p2).sub(p1).scl(t).scl(2));
    }

// Bezier.java : public static T cubic(T out, float t, T p0, T p1, T p2, T p3, T tmp)
    public static <T extends Vector<T>> T cubic(final T out, final float t, final T p0, final T p1, final T p2, final T p3,
                                                final T tmp){
        // B3(t) = (1-t) * (1-t) * (1-t) * p0 + 3 * (1-t) * (1-t) * t * p1 + 3 * (1-t) * t * t * p2 + t * t * t * p3
        final float dt = 1f - t;
        final float dt2 = dt * dt;
        final float t2 = t * t;
        return out.set(p0).scl(dt2 * dt).add(tmp.set(p1).scl(3 * dt2 * t)).add(tmp.set(p2).scl(3 * dt * t2))
        .add(tmp.set(p3).scl(t2 * t));
    }

// Bezier.java : public static T cubicDerivative(T out, float t, T p0, T p1, T p2, T p3, T tmp)
    public static <T extends Vector<T>> T cubicDerivative(final T out, final float t, final T p0, final T p1, final T p2,
                                                          final T p3, final T tmp){
        // B3'(t) = 3 * (1-t) * (1-t) * (p1 - p0) + 6 * (1 - t) * t * (p2 - p1) + 3 * t * t * (p3 - p2)
        final float dt = 1f - t;
        final float dt2 = dt * dt;
        final float t2 = t * t;
        return out.set(p1).sub(p0).scl(dt2 * 3).add(tmp.set(p2).sub(p1).scl(dt * t * 6)).add(tmp.set(p3).sub(p2).scl(t2 * 3));
    }

// ConvexHull.java : private float ccw(float p3x, float p3y)
    private float ccw(float p3x, float p3y){
        FloatSeq hull = this.hull;
        int size = hull.size;
        float p1x = hull.get(size - 4);
        float p1y = hull.get(size - 3);
        float p2x = hull.get(size - 2);
        float p2y = hull.peek();
        return (p2x - p1x) * (p3y - p1y) - (p2y - p1y) * (p3x - p1x);
    }

// Ellipse.java : public float circumference()
    public float circumference(){
        float a = this.width / 2;
        float b = this.height / 2;
        if(a * 3 > b || b * 3 > a){
            // If one dimension is three times as long as the other...
            return (float)(Mathf.PI * ((3 * (a + b)) - Math.sqrt((3 * a + b) * (a + 3 * b))));
        }else{
            // We can use the simpler approximation, then
            return (float)(Mathf.PI2 * Math.sqrt((a * a + b * b) / 2));
        }
    }

// Intersector.java : public static Vec2 nearestSegmentPoint(Vec2 start, Vec2 end, Vec2 point, Vec2 nearest) / public static Vec2 nearestSegmentPoint(float startX, float startY, float endX, float endY, float pointX, float pointY, Vec2 nearest)
    public static Vec2 nearestSegmentPoint(Vec2 start, Vec2 end, Vec2 point, Vec2 nearest){
        float length2 = start.dst2(end);
        if(length2 == 0) return nearest.set(start);
        float t = ((point.x - start.x) * (end.x - start.x) + (point.y - start.y) * (end.y - start.y)) / length2;
        if(t < 0) return nearest.set(start);
        if(t > 1) return nearest.set(end);
        return nearest.set(start.x + t * (end.x - start.x), start.y + t * (end.y - start.y));
    }

// Intersector.java : public static float intersectRayRay(Vec2 start1, Vec2 direction1, Vec2 start2, Vec2 direction2)
    public static float intersectRayRay(Vec2 start1, Vec2 direction1, Vec2 start2, Vec2 direction2){
        float difx = start2.x - start1.x;
        float dify = start2.y - start1.y;
        float d1xd2 = direction1.x * direction2.y - direction1.y * direction2.x;
        if(d1xd2 == 0.0f){
            return Float.POSITIVE_INFINITY; // collinear
        }
        float d2sx = direction2.x / d1xd2;
        float d2sy = direction2.y / d1xd2;
        return difx * d2sy - dify * d2sx;
    }
