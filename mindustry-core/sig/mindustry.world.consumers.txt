Consume.java
Java class (abstract)
// An abstract class that defines a type of resource that a block can consume.
fields:
  public boolean optional  // If true, this consumer will not influence consumer validity.
  public boolean booster  // If true, this consumer will be displayed as a boost input.
  public boolean update  // If false, this consumer will still be checked, but it will need to updated manually.
  public Floatf<Building> multiplier  // Multiplier for costs.
methods:
  public void apply(Block block)  // Apply extra filters to a block.
  public Consume optional(boolean optional, boolean boost)
  public Consume boost()
  public Consume update(boolean update) / public void update(Building build)
  public boolean ignore()  // @return if true, this consumer will be ignored in the consumer list (no updates or valid() checks)
  public void build(Building build, Table table)
  public void trigger(Building build)  // Called when a consumption is triggered manually.
  public float efficiency(Building build)  // @return [0, 1] efficiency multiplier based on input.
  public float efficiencyMultiplier(Building build)  // @return multiplier for efficiency - this can be above 1.
  public void display(Stats stats)
