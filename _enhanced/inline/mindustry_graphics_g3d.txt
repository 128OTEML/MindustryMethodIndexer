// MeshBuilder.java : private static Mesh end(Mesh mesh)
    private static Mesh end(Mesh mesh){
        mesh.getVerticesBuffer().limit(mesh.getVerticesBuffer().position());
        if(mesh.getNumIndices() > 0){
            mesh.getIndicesBuffer().limit(mesh.getIndicesBuffer().position());
        }

        return mesh;
    }

// MeshBuilder.java : private static Vec3 normal(Vec3 v1, Vec3 v2, Vec3 v3, Vec3 out) / private static void normal(float v1x, float v1y, float v1z, float v2x, float v2y, float v2z, float v3x, float v3y, float v3z, Vec3 out)
    private static Vec3 normal(Vec3 v1, Vec3 v2, Vec3 v3, Vec3 out){
        return out.set(v2).sub(v1).crs(v3.x - v1.x, v3.y - v1.y, v3.z - v1.z).nor();
    }

// MeshBuilder.java : private static void verts(Mesh mesh, float[] floats, Vec3 a, float h1, Vec3 b, float h2, Vec3 c, float h3, Vec3 normal, float color, float emissive)
    private static void verts(Mesh mesh, float[] floats, Vec3 a, float h1, Vec3 b, float h2, Vec3 c, float h3, Vec3 normal, float color, float emissive){
        vert(mesh, floats, a.x * h1, a.y * h1, a.z * h1, normal, color, emissive);
        vert(mesh, floats, b.x * h2, b.y * h2, b.z * h2, normal, color, emissive);
        vert(mesh, floats, c.x * h3, c.y * h3, c.z * h3, normal, color, emissive);
    }

// MeshBuilder.java : private static float packNormals(float x, float y, float z)
    private static float packNormals(float x, float y, float z){
        int xs = x < -1f/512f ? 1 : 0;
        int ys = y < -1f/512f ? 1 : 0;
        int zs = z < -1f/512f ? 1 : 0;

        int vi =
        zs << 29 | ((int)(z * 511 + (zs << 9)) & 511) << 20 |
        ys << 19 | ((int)(y * 511 + (ys << 9)) & 511) << 10 |
        xs << 9  | ((int)(x * 511 + (xs << 9)) & 511);
        return Float.intBitsToFloat(vi);
    }

// PlanetGrid.java : static void addCorner(int id, PlanetGrid grid, int t1, int t2, int t3)
    static void addCorner(int id, PlanetGrid grid, int t1, int t2, int t3){
        Corner c = grid.corners[id];
        Ptile[] t = {grid.tiles[t1], grid.tiles[t2], grid.tiles[t3]};
        c.v.set(t[0].v).add(t[1].v).add(t[2].v).nor();
        for(int i = 0; i < 3; i++){
            t[i].corners[pos(t[i], t[(i + 2) % 3])] = c;
            c.tiles[i] = t[i];
        }
    }

// PlanetGrid.java : static int pos(Ptile t, Ptile n) / static int pos(Ptile t, Corner c) / static int pos(Corner c, Corner n)
    static int pos(Ptile t, Ptile n){
        for(int i = 0; i < t.edgeCount; i++){
            if(t.tiles[i] == n) return i;
        }
        return -1;
    }

    }

// PlanetRenderer.java : public void renderOrbit(Planet planet, PlanetParams params)
    public void renderOrbit(Planet planet, PlanetParams params){
        if(planet.parent == null || !planet.visible() || params.uiAlpha <= 0.02f || !planet.drawOrbit) return;

        Vec3 center = planet.parent.position;
        float radius = planet.orbitRadius;
        int points = (int)(radius * 10);
        Angles.circleVectors(points, radius, (cx, cy) -> batch.vertex(Tmp.v32.set(center).add(cx, 0, cy), Pal.gray.write(Tmp.c1).a(params.uiAlpha)));
        batch.flush(Gl.lineLoop);
    }

// PlanetRenderer.java : public void renderSectors(Planet planet, PlanetParams params)
    public void renderSectors(Planet planet, PlanetParams params){
        if(params.uiAlpha <= 0.02f) return;
        planet.renderSectors(batch, cam, params);
    }
