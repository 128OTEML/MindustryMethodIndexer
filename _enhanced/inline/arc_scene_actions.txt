// AfterAction.java : protected boolean delegate(float delta)
    @Override
    protected boolean delegate(float delta){
        Seq<Action> currentActions = target.getActions();
        if(currentActions.size == 1) waitForActions.clear();
        for(int i = waitForActions.size - 1; i >= 0; i--){
            Action action = waitForActions.get(i);
            int index = currentActions.indexOf(action, true);
            if(index == -1) waitForActions.remove(i);
        }
        return waitForActions.size <= 0 && action.act(delta);
    }

    @Override
    protected boolean delegate(float delta){

            if(time < duration) return false;

    }

// DelegateAction.java : public final boolean act(float delta)
    @Override
    public final boolean act(float delta){
        Pool pool = getPool();
        setPool(null); // Ensure this action can't be returned to the pool inside the delegate action.
        try{
            return delegate(delta);
        }finally{
            setPool(pool);
        }
    }

// RepeatAction.java : protected boolean delegate(float delta)
    @Override
    protected boolean delegate(float delta){
        if(executedCount == repeatCount) return true;
        if(action.act(delta)){
            if(finished) return true;
            if(repeatCount > 0) executedCount++;
            if(executedCount == repeatCount) return true;
            if(action != null) action.restart();
        }
        return false;
    }

// RepeatAction.java : public void finish()
    public void finish(){
        finished = true;
    }

// RepeatAction.java : public void restart()
    @Override
    public void restart(){
        super.restart();
        executedCount = 0;
        finished = false;
    }

        Pool pool = getPool();
        setPool(null); // Ensure this action can't be returned to the pool inside the runnable.
        try{
            runnable.run();

        }
