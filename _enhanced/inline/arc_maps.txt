// MapGroupLayer.java : public void invalidateRenderOffset()
    @Override
    public void invalidateRenderOffset(){
        super.invalidateRenderOffset();
        for(int i = 0; i < layers.size; i++){
            MapLayer child = layers.get(i);
            child.invalidateRenderOffset();
        }
    }

// MapLayer.java : public float getRenderOffsetX()
    public float getRenderOffsetX(){
        if(renderOffsetDirty) calculateRenderOffsets();
        return renderOffsetX;
    }

// MapLayer.java : public float getRenderOffsetY()
    public float getRenderOffsetY(){
        if(renderOffsetDirty) calculateRenderOffsets();
        return renderOffsetY;
    }

// MapLayer.java : public void invalidateRenderOffset()
    public void invalidateRenderOffset(){
        renderOffsetDirty = true;
    }

// MapLayer.java : public MapLayer getParent()
    public MapLayer getParent(){
        return parent;
    }

// MapLayer.java : public void setParent(MapLayer parent)
    public void setParent(MapLayer parent){
        if(parent == this) throw new ArcRuntimeException("Can't set self as the parent");
        this.parent = parent;
    }

// MapLayer.java : protected void calculateRenderOffsets()
    protected void calculateRenderOffsets(){
        if(parent != null){
            parent.calculateRenderOffsets();
            renderOffsetX = parent.getRenderOffsetX() + offsetX;
            renderOffsetY = parent.getRenderOffsetY() + offsetY;
        }else{
            renderOffsetX = offsetX;
            renderOffsetY = offsetY;
        }
        renderOffsetDirty = false;
    }

    public Cell getCell(int x, int y){
        if(x < 0 || x >= width || y < 0 || y >= height) return null;
        return cells[x][y];
    }

// TileLayer.java : public void setCell(int x, int y, Cell cell)
    public void setCell(int x, int y, Cell cell){
        if(x < 0 || x >= width || y < 0 || y >= height) return;
        cells[x][y] = cell;
    }

    public MapTile getTile(int x, int y){
        return getCell(x, y) == null ? null : getCell(x, y).tile;
    }
